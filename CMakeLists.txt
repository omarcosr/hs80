cmake_minimum_required(VERSION 3.24)
project(HS80Control VERSION 2.1.0 LANGUAGES C RC)
if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Este recorte usa Windows x64 e MSVC.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(vendor/CUI)
add_library(hs80_common STATIC src/common.c src/protocol.c)
target_include_directories(hs80_common PUBLIC src)
target_compile_definitions(hs80_common PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_link_libraries(hs80_common PUBLIC advapi32 shell32 user32 psapi)
add_executable(hs80 WIN32 src/controller.c src/ui.c src/hid.c src/audio.c src/app.rc src/app.manifest)
set_target_properties(hs80 PROPERTIES OUTPUT_NAME HS80)
target_link_libraries(hs80 PRIVATE hs80_common cui::cui hid setupapi ole32 uuid)
add_executable(hs80_tests tests/protocol_test.c src/protocol.c)
target_include_directories(hs80_tests PRIVATE src)
add_executable(hs80_memory_tests tests/ui_memory_test.c vendor/CUI/src/cui_hs80_memory.c)
target_include_directories(hs80_memory_tests PRIVATE vendor/CUI/include)
enable_testing()
add_test(NAME protocol_and_validation COMMAND hs80_tests)
add_test(NAME ui_memory_lifetime COMMAND hs80_memory_tests)
foreach(t hs80_common hs80 hs80_tests hs80_memory_tests)
 target_compile_options(${t} PRIVATE /W4 /utf-8 /Gy /Gw /O1)
endforeach()
set_property(TARGET hs80_common cui hs80 PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
target_link_options(hs80 PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /MAP)
add_custom_target(dist
 COMMAND "${CMAKE_COMMAND}" -E make_directory "${PROJECT_SOURCE_DIR}/dist/licencas"
 COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:hs80>" "${PROJECT_SOURCE_DIR}/dist/HS80.exe"
 COMMAND "${CMAKE_COMMAND}" -DHS80_ROOT=${PROJECT_SOURCE_DIR} -P "${PROJECT_SOURCE_DIR}/cmake/PrepareDist.cmake"
 DEPENDS hs80 hs80_tests hs80_memory_tests
 COMMENT "Preparando dist com um unico executavel"
 VERBATIM)
